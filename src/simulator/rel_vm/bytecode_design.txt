Instructions:
    IADD dest src1 src2
    ISUB dest src1 src2 // src1-src2
    IMUL dest src1 src2
    IDIV dest src1 src2 // src1/src2
    IABS dest src       // |src|
    IMOD  dest src1 src2 // src1 % src2
    I2F dest src     // Convert int to float

    FADD dest src1 src2
    FSUB dest src1 src2
    FMUL dest src1 src2
    FDIV dest src1 src2
    FABS dest src       // |src|
    FEXP dest src       // e^src
    FLN  dest src       // ln(src)

    CADD dest src1 src2
    CSUB dest src1 src2
    CMUL dest src1 src2
    CDIV dest src1 src2
    CABS dest src       
    CARG dest src   // argument (angle) of complex number
    CEXP dest src       
    CLN  dest src       
    CON  dest src   // take conjuate of complex number

    AND dest src1, src2
    OR dest src1, src2
    NOT dest src   
    XOR dest src1, src2

    FLR dest scr     // convert float to int with floor
    F2C dest src     // convert float to complex
    CRE dest src     // return real part of complex as float
    CIM dest src     // return imaginary part of complex as float

    JMP offset             // unconditional jump 

    IJEQ offset src1 src2 // Jump if equal
    IJNE offset src1 src2 // Jump if not equal
    IJLT offset src1 src2 // jump if less than
    IJGT offset src1 src2 // jump if greater than
    IJLE offset src1 src2 // jump if less than or equal
    IJGE offset src1 src2 // jump if greater than or equal

    FJEQ offset src1 src2 // Jump if equal
    FJNE offset src1 src2 // Jump if not equal
    FJLT offset src1 src2 // jump if less than
    FJGT offset src1 src2 // jump if greater than
    FJLE offset src1 src2 // jump if less than or equal
    FJGE offset src1 src2 // jump if greater than or equal

    CJEQ offset src1 src2
    CJNE offset src1 src2

    IEQ dest src1 src2 // src1 == src2
    INE dest src1 src2 // src1 != src2
    ILT dest src1 src2 // src1 < src2
    IGT dest src1 src2 // src1 > src2
    ILE dest src1 src2 // src1 <= src2
    IGE dest src1 src2 // src1 >= src2

    FEQ dest src1 src2 
    FNE dest src1 src2
    FLT dest src1 src2
    FGT dest src1 src2
    FLE dest src1 src2 
    FGE dest src1 src2 

    CEQ dest src1 src2
    CNE dest src1 src2


    MOV dest, src 
    RET src
    ERR code, src // src is optional depending on ss bits

    all encodings are little-endian

    RND dest // samples a uniformly distributed random variable in range [0,1)
    RNDN dest // samples a normally distributed random variable with mean 0 sd 1

	sources can be literal or a virtual register

    jump offsets are i16 
    error codes are u8

    There are 64 registers, all 64 bits, registers addressed with 8 bits but top 2 ignored

    simulation timestamp put into r0
    relation delay put into r1
    n arguments are put into r2..rn+1

    little endian

    for assembler "#" makes a comment but they must be on their own line 

    all values are encoded in 64 bits
        i64, f64, and complex numbers will be 2 f32s packed into one 64 bit reg.

bits:
    a a f d d d s s

    a a:
        0 0: arithmetic instruction
        0 1: logical / arithmetic instructions
        1 0: jump instruction
        1 1: compare instrcution

    0 x f:
        0: integer arithmetic / logical operations / other math operations
        1: float arithmetic / complex arithmetic

    1 x f: 
        0: integer jumps /comparisons / other operations
        1: float jump / comparisons / other operations



    0 0 0 d d d:
        1 0 1: IMOD
        1 1 0: I2F // to convert int to float (float)
        1 1 1: CJEQ

    0 0 1 d d d:
        1 0 1: FEXP
        1 1 0: FLN
        1 1 1: CJNE

    0 0 f d d d:
        0 0 0: ADD
        0 0 1: SUB
        0 1 0: MUL
        0 1 1: DIV
        1 0 0: ABS



    0 1 0 0 d d:
        0 0: AND
        0 1: OR
        1 0: NOT
        1 1: XOR

    0 1 0 1 d d:
        0 0: FLR // floor
        0 1: F2C // to convert float to complex   
        1 0: CRE // real part of a complex number
        1 1: CIM // imaginary part of a complex number        

    0 1 1 d d d 0 s:
        1 0 0 _ 0: CABS
        1 0 0 _ 1: CARG

    0 1 1 d d d:
        0 0 0: CADD         
        0 0 1: CSUB         
        0 1 0: CMUL         
        0 1 1: CDIV 
        1 0 1: CEXP
        1 1 0: CLN
        1 1 1: CON



    1 0 0 d d d:
        0 0 0: JMP
        0 0 1: MOV

    1 0 1 d d d:
        0 0 0: RET
        0 0 1: ERR

    1 0 f d d d:
        0 1 0: JEQ
        0 1 1: JNE
        1 0 0: JLT
        1 0 1: JLE
        1 1 0: JGT
        1 1 1: JGE



    1 1 0 d d d:
        0 0 0: CEQ
        0 0 1: CNE

    1 1 1 d d d:   
        0 0 0: RND
        0 0 1: RNDN

    1 1 f d d d:
        0 1 0: EQ
        0 1 1: NE
        1 0 0: LT
        1 0 1: LE
        1 1 0: GT
        1 1 1: GE



    s s: source sources
        0 0: reg       reg
        0 1: reg       immediate
        1 0: immediate reg
        1 1: immediate immediate

    s s (for ERR):
        0 0: no src
        0 1: no src
        1 0: src as reg
        1 1: src as immediate
