Instructions:
	MOV dest, src

    IADD dest, src1, src2
    ISUB dest, src1, src2
    IMUL dest, src1, src2
    IDIV dest, src1, src2
	IPOW dest, src1, src2
	INEG dest, src

	FADD dest, src1, src2
    FSUB dest, src1, src2
    FMUL dest, src1, src2
    FDIV dest, src1, src2
	FPOW dest, src1, src2
	FNEG dest, src

    AND dest, src1, src2
    OR dest, src1, src2
	NOT dest, src

    MOD dest, src1, idx // src1 % idx

    JEQ offset, src1, src2 // Jump if equal
    JLT offset, src1, src2 // jump if less than
    JGT offset, src1, src2 // jump if greater than
    JLE offset, src1, src2 // jump if less than or equal
    JGE offset, src1, src2 // jump if greater than or equal
    JMP offset             // unconditional jump

	RND dest // generates a random number in range [0,1)


	sources can be literal or a virtual register
